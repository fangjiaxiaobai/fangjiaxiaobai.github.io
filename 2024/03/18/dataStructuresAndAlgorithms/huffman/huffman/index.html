深入解析 Huffman 编码：原理与应用

Huffman编码，作为一种高效的前缀编码方式，由美国计算机科学家戴维·A·哈夫曼于 1952 年提出。它在数据压缩领域具有重要地位，尤其适用于文本、音频、视频等数据的压缩处理，其核心优势在于能够根据字符出现频率动态生成最优的编码方案。

Huffman编码原理

Huffman编码的基本思想是基于字符出现的频率来构造一种变长编码。对于出现频率高的字符，赋予较短的编码；反之，对出现频率低的字符赋予较长的编码。这种策略确保了整体编码长度最小化，从而实现高效的数据压缩。 具体实现步骤如下：

  • 计算每个字符的出现频率。
  • 将每个字符及对应的频率视为一个节点，构建一颗赫夫曼树（Huffman Tree）。通过不断合并频率最低的两个节点，直到树中只剩下一个根节点为止，这个过程保证了频率越高的字符其路径长度越短。
  • 从根节点到每个字符节点的路径所经过的“左”、“右”分支定义了该字符的二进制编码，频率较高的字符对应较短的编码。

代码实现

huffman.h

#ifndef HFTREE_H
#define HFTREE_H

#include <string>
#include <map>
#include <unordered_map>

/**
 * huffman树的节点
 */
class Node {
public:
    std::string character; //字符串(字符or字符组合)
    int freq;  //频率
    Node *left;  //左子树
    Node *right; // 左子树
    bool isLeaf; // 是否为叶子节点

    Node(std::string ch, int freg);

    Node(std::string ch, int freg, bool isLeaf);

    ~Node();
};

/**
 * 用于入堆时的排序
 */
class Compare {
public:
    bool operator()(Node *a, Node *b);
};

class hfTree {

private:

    void encode(); // huffman编码执行入口

    std::string text;
    Node *root;
    std::unordered_map<std::string, int> frequencyMap;  // 各字符(组合)的频率表
    std::map<std::string, std::string> huffmanCodeMap;  // 编码表

    void calculateFrequency();

    void buildHuffmanTree();

    void generateCodes(Node *node, const std::string &str);

    void deleteTree(Node *node);

public:
    enum class Option {
        SingleChar,
        MultiChar
    };

    Option op;

    hfTree(const std::string &text, Option op);

    ~hfTree();

    std::string getEncodedText();

    std::map<std::string, std::string> getCodingTable();
};

#endif

huffman.cpp

#include "hftree.h"
#include <unordered_map>
#include <vector>
#include <queue>
#include <string>
#include <iostream>

#define DEBUG 1
#define INFO 2
#define OFF 3
#define LOG_LEVEL OFF

using namespace std;


bool contains(std::vector<std::string> container, std::string v) {
    auto it = std::find(container.begin(), container.end(), v);

    if (it != container.end()) {
        return true;
    } else {
        return false;
    }
}

/**
 * Node 构造函数
 * @param ch 字符
 * @param freq 频率
 */
Node::Node(std::string ch, int freq) : character(ch), freq(freq), left(nullptr), right(nullptr), isLeaf(false) {}

/**
 * Node 构造函数
 * @param ch 字符
 * @param freq 频率
 */
Node::Node(std::string ch, int freq, bool isLeaf) : character(ch), freq(freq), left(nullptr), right(nullptr), isLeaf(isLeaf) {}


/**
 * Node 析构函数
 */
Node::~Node() {
}

bool Compare::operator()(Node *a, Node *b) {
    if (a->freq == b->freq) {
        return a->character > b->character;
    } else {
        return a->freq > b->freq;
    }
}


hfTree::hfTree(const std::string &text, Option op) : op(op), root(nullptr), text(text) {
    encode();
}

hfTree::~hfTree() {
    deleteTree(root);
}

void hfTree::deleteTree(Node *node) {
    if (node == nullptr) return;
    deleteTree(node->left);
    deleteTree(node->right);
    delete node;
}

void hfTree::calculateFrequency() {
    if (op == Option::SingleChar) {
        for (char ch: text) {
            std::string key(1, ch);
            frequencyMap[key]++;
        }
    } else {
        std::map<std::string, int> tempFrequencyMap;
        std::vector<std::string> orderHfNodes;

        for (int i = 0; i < text.length(); i++) {
            std::string key(1, text[i]);
            frequencyMap[key]++;
            if (i + 1 < text.length()) {
                key += text[i + 1];
                tempFrequencyMap[key]++;
                orderHfNodes.push_back(key);
            }
        }

        // 选择频率最高的前三个字符组合
        std::vector<std::pair<std::string, int>> vec(tempFrequencyMap.begin(), tempFrequencyMap.end());
        std::sort(vec.begin(), vec.end(), [](const auto &a, const auto &b) {
            return a.second > b.second || (a.second == b.second && a.first < b.first);
        });

        // 如果字符组合不足三个，则全部选择
        int topComboCount = std::min(3, static_cast<int>(vec.size()));

        std::vector<std::string> top3;
        // 更新单字符频率表，减去组合中字符的频率
        for (int i = 0; i < topComboCount; ++i) {
            // 将前三个字符组合添加到频率表中
            frequencyMap[vec[i].first] = vec[i].second;
            top3.push_back(vec[i].first);
            // 减去组合中的单个字符的频率
            frequencyMap[std::string(1, vec[i].first[0])] -= vec[i].second;
            frequencyMap[std::string(1, vec[i].first[1])] -= vec[i].second;
        }

        int i = 0;
        for (; i < orderHfNodes.size() - 2; i++) {
            if (contains(top3, orderHfNodes[i]) &&
                contains(top3, orderHfNodes[i + 1])) {
                if (contains(top3, orderHfNodes[i + 2])) {
                    frequencyMap[orderHfNodes[++i]]--;
                } else {
                    std::string k = orderHfNodes[++i];
                    frequencyMap[k]--;
                    frequencyMap[std::string(1, orderHfNodes[i][1])]++;
                    frequencyMap[std::string(1, orderHfNodes[i][0])]++;
//                    std::cout << i << ":" << k << std::endl;
                }
            }
        }
//        std::cout << "_t:" << frequencyMap[" t"] << " t:" << frequencyMap["t"] << endl;
        if(i==orderHfNodes.size() - 2){
            if (contains(top3, orderHfNodes[i]) &&
                contains(top3, orderHfNodes[i + 1])){
                frequencyMap[orderHfNodes[++i]]--;
                frequencyMap[std::string(1, orderHfNodes[i][1])]++;
                frequencyMap[std::string(1, orderHfNodes[i][0])]++;
            }
        }
#if LOG_LEVEL <= DEBUG
        std::cout << "\n字符对编码:去0前:" << std::endl;
        for (auto p: frequencyMap) {
            std::cout << p.first << "-" << p.second << ",";
        }
        std::cout << std::endl;
#endif

        // 过滤掉v为0的pair
        for (auto it = frequencyMap.begin(); it != frequencyMap.end(); /* no increment here */) {
            if (it->second <= 0) {
                // Erase element by iterator and get the next iterator position
                it = frequencyMap.erase(it);
            } else {
                // Only increment the iterator if not erasing
                ++it;
            }
        }

#if LOG_LEVEL <= DEBUG
        std::cout << "去0后:" << std::endl;
        for (auto p: frequencyMap) {
            std::cout << p.first << "-" << p.second << std::endl;
        }
        std::cout << std::endl;
#endif
    }

#if LOG_LEVEL <= INFO
    std::cout << "\n>>>>>>>>>>>>>> 1、构建字符频率表:" << std::endl;
    for (auto p: frequencyMap) {
        std::cout << p.first << "-" << p.second << std::endl;
    }
    std::cout << "\n" << std::endl;
#endif
}

void hfTree::buildHuffmanTree() {

    std::priority_queue<Node *, std::vector<Node *>, Compare> minHeap;

    for (const auto &pair: frequencyMap) {
        Node *newNode = new Node(pair.first, pair.second, true);
        minHeap.push(newNode);
    }

#if LOG_LEVEL <= DEBUG
    printf("\n 小顶堆构建过程:\n");
#endif
    while (minHeap.size() > 1) {
        Node *left = minHeap.top();
        minHeap.pop();
        Node *right = minHeap.top();
        minHeap.pop();
        auto chs = min(left->character, right->character) + max(left->character, right->character);
        Node *sum = new Node(chs, left->freq + right->freq);
        sum->left = left;
        sum->right = right;
        minHeap.push(sum);
#if LOG_LEVEL <= DEBUG
        printf("left:%s-%d, right:%s-%d, parent:%s-%d\n", (left->character.c_str()), left->freq,
               right->character.c_str(), right->freq,
               sum->character.c_str(), sum->freq);
#endif
    }
#if LOG_LEVEL <= DEBUG
    printf(" 小顶堆构建构建完成\n");
#endif
    root = minHeap.top();
    minHeap.pop();
}


void hfTree::generateCodes(Node *node, const std::string &str) {
    if (node == nullptr) {
        return;
    }

    if (node->isLeaf) {
        huffmanCodeMap[node->character] = str;
    }

    generateCodes(node->left, str + "0");
    generateCodes(node->right, str + "1");
}

void hfTree::encode() {
    calculateFrequency(); // 统计各字符出现的次数(频率)
    buildHuffmanTree();  // 构建haffman树
    generateCodes(root, ""); //对huffman中的字符(组合)
#if LOG_LEVEL == INFO
    std::cout << "\n>>>>>>>>>>>>>> 1、构建字符编码表:" << std::endl;
    for (auto p: huffmanCodeMap) {
        std::cout << p.first << ":" << p.second << std::endl;
    }
    std::cout << "\n" << std::endl;
#endif
}

std::string hfTree::getEncodedText() {
    std::string encodedText;
    for (char ch: text) {
        encodedText += huffmanCodeMap[std::string(1, ch)];
    }
    return encodedText;
}


std::map<std::string, std::string> hfTree::getCodingTable() {
    return huffmanCodeMap;
}

Huffman 编码的应用

  • 数据压缩：在文本文件、音频流和视频流等数据压缩场景中广泛应用，如 JPEG 图像格式、MPEG 视频格式以及 PNG 图像格式的部分压缩流程都使用到了Huffman编码。
  • 通信传输：在网络通信领域，Huffman编码能有效减少数据传输量，提高带宽利用率，特别是在带宽资源有限的情况下，其效果尤为显著。
  • 存储优化：在磁盘存储或内存管理中，通过Huffman编码可以节省存储空间，提升存储效率。

总结

Huffman编码利用信息熵理论，将数据编码为更短的形式，实现了数据的有效压缩，是现代信息技术中的关键技术之一。尽管随着技术发展出现了许多更为复杂的压缩算法，但Huffman编码因其简洁高效、易于实现的特性，在很多实际应用场景中仍占据着不可替代的地位。同时，理解并掌握Huffman编码的思想和方法，也有助于我们深入理解数据结构、算法设计等相关领域的基础知识。
